@startuml
!includeurl https://raw.githubusercontent.com/plantuml-stdlib/C4-PlantUML/master/C4_Container.puml

Person(user, "User")

System_Ext(min_hacienda, "Ministerio de Hacienda")
System_Ext(vertex_ai, "Vertex AI")
System_Ext(ocr_service, "OCR Service")
System_Ext(ai_api, "AI API")

System_Boundary(system, "DUA Processing System") {

    Container(ui, "Frontend", "Web App", "Upload docs and view DUA")

    Container(api, "API", "REST API", "Entry point")

    Container(contracts, "Contracts", "DTOs + Interfaces", "Defines all system contracts")

    Container(acl, "Anti-Corruption Layer", "Adapters", "Implements translation using contracts")

    Container(storage, "Storage", "GCS", "Stores files")
    Container(db, "Database", "PostgreSQL", "Stores metadata")
    Container(vector_db, "Vector DB", "Vector Store", "Stores embeddings")

    ' ===== BACKEND CORE =====
    Container_Boundary(backend, "Backend Core") {

        Container(model, "Model", "Domain", "Business entities")

        Container(controller, "Controllers", "Layer", "Handle requests")

        Container(services, "Services", "Layer", "Business logic")

        Container(repositories, "Repositories", "Layer", "Data access")
    }
}

' ===== USER FLOW =====
Rel(user, ui, "Uses")
Rel(ui, api, "HTTPS")

' ===== API ↔ CONTRACTS =====
Rel(api, contracts, "Uses DTOs")

' ===== CONTROLLER VIA CONTRACTS =====
Rel(controller, contracts, "Uses request/response DTOs")
Rel(contracts, services, "Service interfaces (ports)")

' ===== SERVICES =====
Rel(services, repositories, "Uses")
Rel(repositories, db, "Reads/Writes")

Rel(services, vector_db, "Embeddings")
Rel(services, storage, "Reads files")

Rel(services, model, "Uses domain")

' ===== ACL (SOLO CONTRACTS) =====
Rel(services, contracts, "Uses external service contracts")
Rel(contracts, acl, "Implements adapters")

' ===== EXTERNAL SYSTEMS (LLAMADOS DESDE SERVICES) =====
Rel(services, vertex_ai, "Uses (via contracts)")
Rel(services, ocr_service, "Uses (via contracts)")
Rel(services, ai_api, "Uses (via contracts)")
Rel(services, min_hacienda, "Uses (via contracts)")

' ===== RESPONSE =====
Rel(services, contracts, "Returns DTOs")
Rel(api, ui, "Returns DUA")

@enduml
