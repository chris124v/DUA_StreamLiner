@startuml
!includeurl https://raw.githubusercontent.com/plantuml-stdlib/C4-PlantUML/master/C4_Container.puml

Person(user, "User")

System_Ext(min_hacienda, "Ministerio de Hacienda")
System_Ext(vertex_ai, "Vertex AI")
System_Ext(ocr_service, "OCR Service")
System_Ext(ai_api, "AI API")

System_Boundary(system, "DUA Processing System") {

    ' ===== PRESENTATION =====
    Container(ui, "Frontend", "Web App", "User interface")

    ' ===== APPLICATION LAYER =====
    System_Boundary(app_layer, "Application Layer") {

        Container(api, "Backend API", "REST API", "Orchestrates workflow")

        Container(dua_generation, "DUA Generation Service", "Service", "Coordinates DUA creation")

    }

    ' ===== DOMAIN LAYER =====
    System_Boundary(domain_layer, "Domain Layer") {

        Container(doc_processing, "Document Processing", "Service", "Parsing, extraction, embeddings")

        Container(template_mgmt, "Template Management", "Service", "Template versioning and structure")

        Container(mapping_domain, "DUA Mapping Domain", "Service", "Business rules for mapping")

    }

    ' ===== INFRASTRUCTURE =====
    System_Boundary(infra_layer, "Infrastructure Layer") {

        Container(storage, "File Storage", "GCS", "Stores files")

        Container(db, "Database", "PostgreSQL", "Stores metadata")

        Container(vector_db, "Vector Store", "Vector DB", "Embeddings")

    }

}

' ===== RELATIONS =====
Rel(user, ui, "Uses")

Rel(ui, api, "HTTPS")

Rel(api, dua_generation, "Triggers")

Rel(dua_generation, doc_processing, "Process documents")
Rel(dua_generation, template_mgmt, "Uses template")
Rel(dua_generation, mapping_domain, "Maps data")

Rel(doc_processing, vector_db, "Stores embeddings")
Rel(doc_processing, storage, "Reads files")

Rel(template_mgmt, min_hacienda, "Fetch template")

Rel(mapping_domain, ai_api, "AI mapping")

Rel(doc_processing, vertex_ai, "Classification")
Rel(doc_processing, ocr_service, "OCR")

Rel(api, db, "Stores metadata")

Rel(api, ui, "Returns DUA")

@enduml
