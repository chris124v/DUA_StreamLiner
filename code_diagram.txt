@startuml
!includeurl https://raw.githubusercontent.com/plantuml-stdlib/C4-PlantUML/master/C4_Component.puml

Container_Boundary(processor, "Document Processor") {

    Component(file_handler, "FileHandler", "Service", "Receives files")

    Component(type_detector, "FileTypeDetector", "Service", "Detects file type")

    Component(thematic_classifier, "ThematicClassifier", "Adapter", "Calls Vertex AI")

    Component(block_parser, "BlockParser", "Service", "Splits documents into blocks")

    Component(excel_parser, "ExcelParser", "Service", "Parses Excel")

    Component(pdf_parser, "PDFParser", "Service", "Parses PDF")

    Component(word_parser, "WordParser", "Service", "Parses Word")

    Component(ocr_adapter, "OCRAdapter", "Adapter", "Handles OCR")

    Component(embedding_generator, "EmbeddingGenerator", "Service", "Generates embeddings")

    Component(hash_service, "HashService", "Service", "Generates hashes")

    Component(vector_indexer, "VectorIndexer", "Service", "Stores/searches embeddings")

    Component(similarity_search, "SimilaritySearch", "Service", "Finds similar blocks")

    Component(field_extractor, "FieldExtractor", "AI Service", "Extracts customs fields")

    Component(validation_service, "ValidationService", "Service", "Validates data")

    Component(mapping_service, "DUAMappingService", "Service", "Prepares data for AI mapping")

    ' ===== TEMPLATE MANAGEMENT =====
    Component(template_fetcher, "TemplateFetcher", "Service", "Fetches official template")

    Component(template_repository, "TemplateRepository", "Storage", "Stores current template")

    Component(template_comparator, "TemplateComparator", "Service", "Compares templates")

    Component(template_parser, "TemplateParser", "Service", "Splits template")

    Component(template_updater, "TemplateUpdater", "Service", "Updates changed blocks")

}

' ===== EXTERNAL AI =====
System_Ext(ai_api, "AI API", "Determines DUA mapping and confidence")

' ===== FILE FLOW =====
Rel(file_handler, type_detector, "Detect type")

Rel(type_detector, word_parser, "Word")
Rel(type_detector, pdf_parser, "PDF")
Rel(type_detector, excel_parser, "Excel")
Rel(type_detector, ocr_adapter, "Image")

Rel(word_parser, block_parser, "Content")
Rel(pdf_parser, block_parser, "Content")
Rel(excel_parser, block_parser, "Content")
Rel(ocr_adapter, block_parser, "OCR text")

Rel(block_parser, thematic_classifier, "Classify")

Rel(block_parser, embedding_generator, "Embeddings")
Rel(block_parser, hash_service, "Hash")

Rel(embedding_generator, vector_indexer, "Store")
Rel(hash_service, vector_indexer, "Check duplicates")

Rel(block_parser, field_extractor, "Extract fields")

Rel(field_extractor, validation_service, "Validate")

Rel(validation_service, similarity_search, "Valid data")

Rel(similarity_search, vector_indexer, "Search")

Rel(similarity_search, mapping_service, "Top matches")

' ===== AI MAPPING (CORREGIDO) =====
Rel(mapping_service, ai_api, "Send candidate blocks")
Rel(ai_api, mapping_service, "Returns DUA section + confidence")

Rel(template_repository, mapping_service, "Provides template structure")

' ===== TEMPLATE FLOW =====
Rel(template_fetcher, template_comparator, "Official template")
Rel(template_repository, template_comparator, "Current template")

Rel(template_comparator, template_parser, "If changes")

Rel(template_parser, embedding_generator, "Embeddings")
Rel(template_parser, hash_service, "Hashes")

Rel(template_parser, template_updater, "Blocks")

Rel(template_updater, template_repository, "Update template")
Rel(template_updater, vector_indexer, "Update embeddings")

@enduml
